package com.unity.androidnotifications;

import android.app.Activity;
import android.content.ActivityNotFoundException;
import android.content.Intent;
import android.os.Bundle;
import android.speech.RecognizerIntent;
import com.unity3d.player.UnityPlayer;
import java.util.ArrayList;
import java.util.Locale;
import org.json.JSONObject;

// The system speech app owns microphone capture and permission prompts.
public final class ReminderVoiceActivity extends Activity {
    public static void launch(Activity activity, String target) {
        activity.runOnUiThread(() -> {
            Intent intent = new Intent(activity, ReminderVoiceActivity.class);
            intent.putExtra("target", target);
            try { activity.startActivity(intent); }
            catch (Exception error) {
                send(target, "error", "Android could not open voice entry. Rebuild and reinstall the latest APK, then try again.");
            }
        });
    }

    @Override public void onCreate(Bundle state) {
        super.onCreate(state);
        if (state != null) return;
        Intent request = new Intent(RecognizerIntent.ACTION_RECOGNIZE_SPEECH);
        request.putExtra(RecognizerIntent.EXTRA_LANGUAGE_MODEL, RecognizerIntent.LANGUAGE_MODEL_FREE_FORM);
        request.putExtra(RecognizerIntent.EXTRA_LANGUAGE, Locale.getDefault().toLanguageTag());
        request.putExtra(RecognizerIntent.EXTRA_MAX_RESULTS, 1);
        request.putExtra(RecognizerIntent.EXTRA_PROMPT, "Say what you want to remember");
        try { startActivityForResult(request, 1); }
        catch (ActivityNotFoundException error) {
            reply("error", "No speech recognition app is available. You can still type your reminder.");
        }
        catch (Exception error) { reply("error", "Voice entry could not start. You can still type your reminder."); }
    }

    @Override protected void onActivityResult(int request, int result, Intent data) {
        super.onActivityResult(request, result, data);
        if (request != 1) return;
        if (result != RESULT_OK) { reply("cancel", ""); return; }
        ArrayList<String> words = data == null ? null : data.getStringArrayListExtra(RecognizerIntent.EXTRA_RESULTS);
        if (words == null || words.isEmpty() || words.get(0).trim().isEmpty())
            reply("error", "No speech was recognized. Try again or type your reminder.");
        else reply("ok", words.get(0).trim());
    }

    private void reply(String status, String text) {
        send(getIntent().getStringExtra("target"), status, text);
        finish();
    }

    private static void send(String target, String status, String text) {
        try {
            JSONObject message = new JSONObject();
            message.put("status", status);
            message.put("text", text);
            UnityPlayer.UnitySendMessage(target, "OnVoiceResult", message.toString());
        } catch (Exception ignored) { }
    }
}
